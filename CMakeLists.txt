cmake_minimum_required(VERSION "3.24")
project("gles-template" VERSION "0.0.1" LANGUAGES CXX)

include("FetchContent")

add_link_options("-Wl,--as-needed")

set(CMAKE_CXX_STANDARD "20")
set(CMAKE_CXX_STANDARD_REQUIRED TRUE)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
set(CMAKE_EXPORT_COMPILE_COMMANDS TRUE)

set(CMAKE_SHARED_LINKER_FLAGS_RELEASE "${CMAKE_SHARED_LINKER_FLAGS_RELEASE} -s")

FetchContent_Declare("sdl3"
  GIT_REPOSITORY "https://github.com/libsdl-org/SDL"
  GIT_TAG "release-3.4.18"
  GIT_SHALLOW TRUE
  GIT_PROGRESS TRUE
)

set(SDL_AUDIO FALSE CACHE BOOL "" FORCE)
set(SDL_CAMERA FALSE CACHE BOOL "" FORCE)
set(SDL_HIDAPI FALSE CACHE BOOL "" FORCE)

set(SDL_ANDROID_JAR TRUE CACHE BOOL "" FORCE)
set(SDL_TESTS FALSE CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY FALSE CACHE BOOL "" FORCE)

set(SDL_SHARED FALSE CACHE BOOL "" FORCE)
set(SDL_STATIC TRUE CACHE BOOL "" FORCE)

FetchContent_MakeAvailable("sdl3")

add_subdirectory("app/src")
